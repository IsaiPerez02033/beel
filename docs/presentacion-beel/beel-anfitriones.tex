% ============ QUÉ ES BEEL ============
\section{\textquestiondown{}Qué es Beel?}
Beel es una plataforma mexicana para publicar y reservar hospedajes ---casas,
departamentos, villas y cabañas---. Conectamos a \textbf{anfitriones locales} con
\textbf{viajeros} que buscan lugares auténticos, con pagos seguros y comunicación
directa. Hecha para México, sin intermediarios corporativos.

\vspace{5pt}
\begin{tcolorbox}[enhanced, colback=beelgreen, colframe=beelgreen, arc=8pt,
  boxrule=0pt, width=\linewidth, halign=center, top=7pt, bottom=7pt]
  {\color{white}\Large\bfseries Recibes el 100\% de tu precio}\\[2pt]
  {\color{white}\small Beel \textbf{no cobra comisión} al anfitrión. La tarifa de servicio
  ($\approx$6\%) la paga el huésped, no tú.}
\end{tcolorbox}

% ============ BENEFICIOS ============
\section{\textquestiondown{}Por qué publicar en Beel?}
\begin{benefit}{Ganas más: recibes el 100\%}
Otras plataformas descuentan al anfitrión del 10\% al 20\%. En Beel recibes el
100\% del precio que defines. La tarifa de servicio la cubre el huésped.
\end{benefit}
\begin{benefit}{Pagos protegidos}
El dinero del huésped se retiene de forma segura y se libera a tu cuenta una vez
confirmada la estancia. Cobras por transferencia a tu CLABE.
\end{benefit}
\begin{benefit}{Huéspedes verificados}
Verificamos teléfono e identidad (INE o pasaporte) de los usuarios, para que
recibas gente real y confiable.
\end{benefit}
\begin{benefit}{Comunicación directa}
Chatea con tus huéspedes dentro de la app: dudas, llegada y detalles de la
reserva, todo en un solo lugar.
\end{benefit}
\begin{benefit}{Publicar es rápido y gratis}
Subes fotos, descripción, ubicación y precio en minutos. Tu anuncio incluye mapa,
reglas de la casa y política de cancelación.
\end{benefit}

% ============ FISCAL ============
\section{Precios e impuestos (importante)}
\begin{tcolorbox}[enhanced, colback=beelsand, colframe=beelsand, arc=6pt,
  boxrule=0pt, left=10pt, right=10pt, top=7pt, bottom=7pt]
\small
$\bullet$~ El precio por noche que defines es el \textbf{monto que recibirás}; se
considera con \textbf{IVA incluido}.\\[2pt]
$\bullet$~ \textbf{Cada anfitrión es responsable de su propia facturación e
impuestos} (ISR e IVA de su hospedaje), igual que en otras plataformas.\\[2pt]
$\bullet$~ Beel emite su propia factura por la tarifa de servicio.\\[2pt]
$\bullet$~ Te recomendamos contar con tu \textbf{RFC} y, si tienes dudas,
apoyarte en un contador.
\end{tcolorbox}

% ============ CÓMO FUNCIONA ============
\section{\textquestiondown{}Cómo funciona?}
\paso{1.}{Creas tu cuenta y verificas tu identidad (una sola vez).}
\paso{2.}{Publicas tu propiedad: fotos, ubicación, precio y reglas.}
\paso{3.}{Recibes solicitudes de reserva y chateas con los huéspedes.}
\paso{4.}{Se completa la estancia y recibes tu pago en tu cuenta bancaria.}

% ============ FAQ ============
\section{Preguntas frecuentes}
\faq{\textquestiondown{}Cuánto cuesta publicar?}{Nada. Publicar es gratis y Beel no te cobra comisión.}
\faq{\textquestiondown{}Cuánto recibo por cada reserva?}{El 100\% del precio que defines. La tarifa de servicio la paga el huésped.}
\faq{\textquestiondown{}Cómo y cuándo me pagan?}{Por transferencia a tu CLABE, una vez confirmada la estancia.}
\faq{\textquestiondown{}Quién se hace cargo de los impuestos?}{Cada anfitrión es responsable de su propia facturación e impuestos. Beel solo factura su tarifa de servicio.}
\faq{\textquestiondown{}Qué pasa con las cancelaciones?}{Tú eliges la política (flexible, moderada o estricta). Los reembolsos se aplican según esa política.}
\faq{\textquestiondown{}Cómo me protege Beel?}{Pagos en garantía (el dinero se libera tras la estancia) y verificación de identidad de los huéspedes.}
